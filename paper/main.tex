\subsection{Skill Self-Evolution Pipeline}
\label{sec:skill_evolution}

\subsubsection{Think-Mode Trace Collection}
\todo{Given training problems, use model in think mode (extended reasoning) to generate solution traces. Verify correctness by executing generated code and matching answers. Retain only successful traces.}

\subsubsection{Procedural Skill Distillation}
\todo{From each verified trace, distill a compact skill consisting of 4 components:
\begin{itemize}
    \item \textbf{Solving Procedure}: Generalized step-by-step approach
    \item \textbf{Worked Example}: Compact arithmetic template (150-250 words)
    \item \textbf{Code Template}: Reusable Python code pattern (max 30 lines)
    \item \textbf{Retrieval Keywords}: 10-20 domain terms for BM25 matching
\end{itemize}
}

\subsubsection{Quality-Aware Skill Curation}
\label{sec:curation}

\paragraph{Leave-One-Out (LOO) Filtering.}
\todo{For each training problem, run inference with skill retrieval but excluding the skill derived from itself. Compare with no-skill baseline. Compute per-skill NetScore = HelpRate - HurtRate. Remove harmful skills (NetScore < 0) and useless high-frequency skills.}

\paragraph{Embedding-Based Merging.}
\todo{Cluster skills by semantic similarity (BGE embeddings, cosine > threshold). Type-aware thresholds (e.g., TSP: 0.92, StandardLP: 0.85). Keep highest-NetScore representative per cluster. Merge keywords for broader coverage.}

\subsection{Skill-Augmented Inference}
\label{sec:inference}

\todo{At test time: (1) Structural-hybrid retrieval (structural pre-filter + BM25 + embedding, RRF fusion) to find top-k skills; (2) Inject skills into prompt; (3) NoThink code generation guided by retrieved skills.}

\subsection{Skill Internalization via SFT and GRPO}
\label{sec:internalization}

\subsubsection{Stage 1: Supervised Fine-Tuning}
\todo{SFT on skill-augmented correct solutions. The model learns to follow structured procedures without explicit skill presence in prompt. Training data: problems paired with skill-guided correct code solutions.}

\subsubsection{Stage 2: GRPO with Curriculum Learning}
\todo{Group Relative Policy Optimization with binary reward (correct/incorrect code execution). Curriculum: Easy -> Medium -> Hard problems. The model internalizes procedural knowledge through outcome-based RL.}

\begin{equation}
    R(q, \hat{y}) = \begin{cases}
        +1 & \text{if code executes correctly and answer matches} \\
        0 & \text{otherwise}
    \end{cases}
\end{equation}
